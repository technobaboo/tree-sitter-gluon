import "spatial.gluon"
import "input.gluon" as input

/// An XR node
interface Node {
    set_position(x: f32, y: f32, z: f32)
    get_transform() -> (transform: Transform)
    destroy()
}

/// A spatial transform
struct Transform {
    position: spatial::Vec3,
    rotation: spatial::Quat,
    scale: spatial::Vec3,
}

/// Input event types
enum InputEvent {
    /// A pointer ray
    Pointer {
        origin: spatial::Vec3,
        direction: spatial::Vec3,
    },
    /// Hand tracking data
    Hand {
        joints: Vec<spatial::Vec3>,
    },
    /// No input
    None,
}
